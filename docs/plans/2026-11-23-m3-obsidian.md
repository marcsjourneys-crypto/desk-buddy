**Goal:** Buddy can save dictated notes and ideas into its own Obsidian vault and read a day's notes back. This uses Buddy's first Claude tools.

**Architecture:**
- A `tools/` package:
  - `registry.py`: `Tool`, `ToolRegistry`, `ToolError`.
  - `notes.py`: `NoteVault` plus the two tool definitions.
- `ClaudeChat.stream_reply` gains a streaming tool loop of up to 3 rounds, with `tool_choice: none` on the last round. The system prompt is rebuilt per request so it can carry today's date.
- The voice loop is unchanged: it still only receives text.

**Tech stack:** Python 3.12 standard library (pathlib, datetime, re, dataclasses) and the existing `anthropic` SDK. No new dependencies.

**Design:** `docs/plans/2026-11-23-m3-obsidian-design.md`

```python
"""Tools Claude can call: plain functions with a strict JSON schema, run by name."""

from __future__ import annotations

import logging
from collections.abc import Callable, Iterable, Mapping
from dataclasses import dataclass
from typing import Any

log = logging.getLogger(__name__)


class ToolError(Exception):
    """A tool couldn't do what was asked. The message is sent back to Claude."""


@dataclass(frozen=True)
class Tool:
    name: str
    description: str
    input_schema: dict[str, Any]  # strict: additionalProperties false, every property required
    func: Callable[..., str]

    def schema(self) -> dict[str, Any]:
        return {
            "name": self.name,
            "description": self.description,
            "input_schema": self.input_schema,
            "strict": True,
        }


@dataclass(frozen=True)
class ToolOutcome:
    text: str
    is_error: bool = False


class ToolRegistry:
    """The one place tools are registered. run() never raises."""

    def __init__(self, tools: Iterable[Tool] = ()) -> None:
        self._tools: dict[str, Tool] = {}
        for tool in tools:
            if tool.name in self._tools:
                raise ValueError(f"duplicate tool name {tool.name!r}")
            self._tools[tool.name] = tool

    def schemas(self) -> list[dict[str, Any]]:
        return [tool.schema() for tool in self._tools.values()]

    def run(self, name: str, args: Mapping[str, Any]) -> ToolOutcome:
        tool = self._tools.get(name)
        if tool is None:
            log.warning("Tool %s: unknown", name)
            return ToolOutcome(f"There is no tool called {name!r}.", is_error=True)
        try:
            text = tool.func(**args)
        except ToolError as exc:
            log.info("Tool %s failed: %s", name, exc)
            return ToolOutcome(str(exc), is_error=True)
        except TypeError as exc:  # arguments don't match the function
            log.warning("Tool %s: bad arguments %r (%s)", name, dict(args), exc)
            return ToolOutcome("The tool was called with the wrong arguments.", is_error=True)
        except Exception:
            log.exception("Tool %s crashed", name)
            return ToolOutcome("The tool failed unexpectedly.", is_error=True)
        log.info("Tool %s -> %s", name, text)
        return ToolOutcome(text)

    def __len__(self) -> int:
        return len(self._tools)
```

```python
"""Buddy's Obsidian vault: dated notes and ideas. Only appends and creates, never edits."""

from __future__ import annotations

import re
from collections.abc import Callable
from datetime import date, datetime
from pathlib import Path

from desk_buddy.tools.registry import ToolError

KINDS = ("note", "idea")
MAX_TITLE = 60
_UNSAFE = re.compile(r'[<>:"/\\|?*\x00-\x1f]')


def safe_title(raw: str) -> str:
    """A title that is safe as part of a Windows filename (may be empty)."""
    title = " ".join(_UNSAFE.sub(" ", raw).split())
    title = re.sub(r"\.{2,}", "", title).strip(" .")
    return title[:MAX_TITLE].rstrip(" .")


def day_heading(day: date) -> str:
    return f"{day:%A} {day.day} {day:%B %Y}"


class NoteVault:
    def __init__(
        self,
        root: Path,
        *,
        daily_folder: str = "Daily",
        ideas_folder: str = "Ideas",
        now: Callable[[], datetime] = datetime.now,
    ) -> None:
        self._root = Path(root).resolve()
        self._daily = daily_folder
        self._ideas = ideas_folder
        self._now = now

    # --- take_note ---

    def take_note(self, text: str, kind: str, title: str) -> str:
        text = " ".join(text.split())
        if not text:
            raise ToolError("There's nothing to note.")
        if kind not in KINDS:
            raise ToolError(f"kind must be one of {', '.join(KINDS)}.")
        now = self._now()
        if kind == "note":
            return self._append_note(now, text)
        return self._create_idea(now, text, title)

    def _append_note(self, now: datetime, text: str) -> str:
        name = f"{now.date().isoformat()}.md"
        path = self._inside(self._daily, name)
        path.parent.mkdir(parents=True, exist_ok=True)
        prefix = ""
        if not path.exists():
            prefix = f"# {day_heading(now.date())}\n\n"
        elif not path.read_text(encoding="utf-8").endswith("\n"):
            prefix = "\n"
        with path.open("a", encoding="utf-8", newline="\n") as f:
            f.write(f"{prefix}- {now:%H:%M} {text}\n")
        return f"Saved to {self._daily}/{name}"

    def _create_idea(self, now: datetime, text: str, title: str) -> str:
        title = safe_title(title) or safe_title(" ".join(text.split()[:6])) or "Idea"
        day = now.date().isoformat()
        body = f"---\ndate: {day}\ntags: [idea, buddy]\n---\n# {title}\n\n{text}\n"
        for n in range(1, 1000):
            name = f"{day} {title}.md" if n == 1 else f"{day} {title} {n}.md"
            path = self._inside(self._ideas, name)
            path.parent.mkdir(parents=True, exist_ok=True)
            try:
                with path.open("x", encoding="utf-8", newline="\n") as f:
                    f.write(body)
            except FileExistsError:
                continue
            return f"Saved idea to {self._ideas}/{name}"
        raise ToolError("Too many ideas with that title today.")

    # --- paths ---

    def _inside(self, folder: str, name: str) -> Path:
        path = (self._root / folder / name).resolve()
        if not path.is_relative_to(self._root):
            raise ToolError("That would be outside the notes folder.")
        return path
```

    def read_notes(self, date: str) -> str:
        try:
            day = datetime.strptime(date, "%Y-%m-%d").date()
        except ValueError:
            raise ToolError("date must look like 2026-11-23.") from None
        iso = day.isoformat()
        notes: list[str] = []
        daily = self._inside(self._daily, f"{iso}.md")
        if daily.is_file():
            notes = [
                line for line in daily.read_text(encoding="utf-8").splitlines()
                if line.startswith("- ")
            ]
        ideas: list[str] = []
        ideas_dir = self._inside(self._ideas, "")
        if ideas_dir.is_dir():
            for path in sorted(ideas_dir.glob(f"{iso} *.md")):
                title = path.stem[len(iso) + 1 :]
                ideas.append(f"- {title}: {_idea_body(path.read_text(encoding='utf-8'))}")
        if not notes and not ideas:
            return f"Nothing noted on {day_heading(day)}."
        parts = [f"Notes for {day_heading(day)}:", *notes]
        if ideas:
            parts += ["Ideas:", *ideas]
        text = "\n".join(parts)
        if len(text) > MAX_READ:
            text = text[: MAX_READ - 20].rstrip() + "\n(and more)"
        return text

def _idea_body(content: str) -> str:
    """The idea's text without frontmatter or heading, on one line."""
    if content.startswith("---\n"):
        end = content.find("\n---\n", 4)
        if end != -1:
            content = content[end + 5 :]
    lines = [ln for ln in content.splitlines() if not ln.startswith("# ")]
    return " ".join(" ".join(lines).split())

def _llm_error(exc: anthropic.APIError) -> LLMError:
    if isinstance(exc, anthropic.RateLimitError):
        return LLMError("rate limited by the API")
    if isinstance(exc, (anthropic.AuthenticationError, anthropic.PermissionDeniedError)):
        return LLMError("API key rejected (check ANTHROPIC_API_KEY)")
    if isinstance(exc, anthropic.APIStatusError):
        # status 200 here means an error event arrived mid-stream; type names it
        return LLMError(f"API error {exc.status_code} ({getattr(exc, 'type', None)})")
    if isinstance(exc, anthropic.APIConnectionError):

    return LLMError(f"unexpected API error ({type(exc).__name__})")

```python
class ClaudeChat:
    """Streaming chat with a tool loop. The caller owns the history (see brain.memory)."""

    def __init__(
        self,
        client: Any,
        *,
        model: str,
        effort: str,
        system: str | Callable[[], str],
        tools: ToolRegistry | None = None,
    ) -> None:
        # client: anthropic.AsyncAnthropic (typed Any so tests can pass a fake)
        self._client = client
        self._model = model
        self._effort = effort
        self._system = system if callable(system) else (lambda: system)
        self._tools = tools or ToolRegistry()

    async def stream_reply(
        self, history: Sequence[Mapping[str, str]], user_text: str
    ) -> AsyncGenerator[str, None]:
        """Yield text deltas as they arrive, running tools between rounds.

        Raises LLMError (any API or network failure) or LLMRefusal. Either can arrive
        after some text was already yielded, so callers must handle a partial reply.
        """
        messages: list[Any] = [*history, {"role": "user", "content": user_text}]
        spoke = False
        for round_no in range(1, MAX_TOOL_ROUNDS + 1):
            params: dict[str, Any] = {
                "model": self._model,
                "max_tokens": MAX_TOKENS,
                "system": self._system(),
                "output_config": {"effort": self._effort},
                "messages": messages,
            }
            if self._tools:
                params["tools"] = self._tools.schemas()
                if round_no == MAX_TOOL_ROUNDS:
                    params["tool_choice"] = {"type": "none"}
            first = True
            try:
                async with self._client.messages.stream(**params) as stream:
                    async for text in stream.text_stream:
                        if first and spoke:
                            yield "\n"  # keep sentences from separate rounds apart
                        first = False
                        spoke = True
                        yield text
                    message = await stream.get_final_message()
            except anthropic.APIError as exc:
                raise _llm_error(exc) from exc
            if message.stop_reason == "refusal":
                raise LLMRefusal("Claude declined to answer")
            if message.stop_reason == "max_tokens":
                log.warning("reply truncated: stop_reason=max_tokens (MAX_TOKENS=%d)", MAX_TOKENS)
                return
            if message.stop_reason != "tool_use":
                return
            results = [
                await self._run_tool(block)
                for block in message.content
                if block.type == "tool_use"
            ]
            messages = [
                *messages,
                {"role": "assistant", "content": message.content},
                {"role": "user", "content": results},
            ]

    async def _run_tool(self, block: Any) -> dict[str, Any]:
        outcome = await asyncio.to_thread(self._tools.run, block.name, block.input)
        result: dict[str, Any] = {
            "type": "tool_result",
            "tool_use_id": block.id,
            "content": outcome.text,
        }
        if outcome.is_error:
            result["is_error"] = True
        return result
```
